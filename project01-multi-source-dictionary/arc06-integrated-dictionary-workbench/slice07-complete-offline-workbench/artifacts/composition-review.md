fn check_evidence(d: &Dictionary, e: &WorkbenchFormEvidence) {
    assert_eq!(e.status, "Loaded");
    assert!(e.reason.is_none());
    assert_eq!(e.raw.as_ref(), d.occurrence(&e.evidence.occurrence));
    assert_eq!(
        e.capture.as_ref(),
        d.capture(&e.evidence.occurrence.capture)
            .map(|x| &x.metadata)
    );
    let r = d.registry().as_document();
    let snap = r.snapshots.iter().find(|x| {
        x.snapshot
            == e.evidence
                .occurrence
                .capture
                .to_snapshot_identity()
                .unwrap()
    });
    assert_eq!(e.snapshot_attribution.as_ref(), snap);
    let oa = r
        .occurrences
        .iter()
        .find(|x| x.occurrence == e.evidence.occurrence.to_record_identity().unwrap());
    assert_eq!(e.occurrence_attribution.as_ref(), oa);
    assert_eq!(
        e.contributor.as_ref(),
        oa.and_then(|x| x.contributor.as_ref())
            .and_then(|id| r.contributors.iter().find(|x| &x.id == id))
    );
    assert_eq!(
        e.distributor.as_ref(),
        snap.and_then(|x| x.distributor.as_ref())
            .and_then(|id| r.distributors.iter().find(|x| &x.id == id))
    );
}

    let collections: BTreeSet<_> = m
        .collections
        .iter()

        .collect();

        .components
        .iter()
        .filter(|x| collections.contains(&x.collection))
        .map(|x| x.id.clone())
        .collect();
    represented.extend(collections);
    represented.extend(components.iter().cloned());
    represented.extend(
        m.releases
            .iter()

    assert_eq!(
        h.extensions,
        m.extensions
            .iter()

            .collect::<Vec<_>>()
    );
}
fn check_exact(
    d: &Dictionary,
    response: &WorkbenchFullResult,
    expected: &BTreeSet<EntityId>,
    count: usize,
) {
    let WorkbenchFullResult::Exact(body) = response else {
        panic!("Exact result")
    };
    assert_eq!(body.totals.occurrences, count);
    assert_eq!(body.totals.source_groups, 4);
    assert!(body.next_cursor.is_none());
    let mut got = Vec::new();
    for g in &body.groups {
        assert_eq!(&g.source, source(d, &g.source.id));
        assert_eq!(g.returned_count, g.hits.len());
        for h in &g.hits {
            check_hit(d, h);
            got.push(h.assertion.id.clone());
        }
    }
    assert_eq!(got.len(), expected.len());
    assert_eq!(ids(got.into_iter()), *expected);
    assert_eq!(
        body.availability.len(),
        d.registry().as_document().sources.len()
    );
    assert_eq!(body.code_notices, d.metadata().code_notices);
    for av in &body.availability {
        let reg = d.registry().as_document();
        let m = d.metadata();
        assert_eq!(&av.source, source(d, &av.source.id));
        assert_eq!(
            av.lineage,
            reg.lineage
                .iter()
                .filter(|x| x.source == av.source.id)
                .cloned()
                .collect::<Vec<_>>()
        );
        assert_eq!(
            av.aliases,
            reg.aliases
                .iter()
                .filter(|x| x.source == av.source.id)
                .cloned()
                .collect::<Vec<_>>()
        );
        let collections: Vec<_> = m
            .collections
            .iter()
            .filter(|x| x.source == av.source.id)
            .cloned()
            .collect();
        assert_eq!(av.collections, collections);
        let collection_ids: BTreeSet<_> = collections.iter().map(|x| &x.id).collect();
        let components: Vec<_> = m
            .components
            .iter()
            .filter(|x| collection_ids.contains(&x.collection))
            .cloned()
            .collect();
        assert_eq!(av.components, components);
        let component_ids: BTreeSet<_> = components.iter().map(|x| &x.id).collect();
        assert_eq!(
            av.releases,
            m.releases
                .iter()
                .filter(|x| component_ids.contains(&x.component))
                .cloned()
                .collect::<Vec<_>>()
        );
        assert_eq!(
            av.owned_head,
            m.owned_heads
                .iter()
                .find(|x| x.source == av.source.id)
                .cloned()
        );
        assert!(av.missing_captures.is_empty());
        assert!(av.selectable);
        assert!(av.reason.is_none());
    }
}

fn check_evidence(d: &Dictionary, e: &WorkbenchFormEvidence) {
    assert_eq!(e.status, "Loaded");
    assert!(e.reason.is_none());
    assert_eq!(e.raw.as_ref(), d.occurrence(&e.evidence.occurrence));
    assert_eq!(
        e.capture.as_ref(),
        d.capture(&e.evidence.occurrence.capture)
            .map(|x| &x.metadata)
    );
    let r = d.registry().as_document();
    let snap = r.snapshots.iter().find(|x| {
        x.snapshot
            == e.evidence
                .occurrence
                .capture
                .to_snapshot_identity()
                .unwrap()
    });
    assert_eq!(e.snapshot_attribution.as_ref(), snap);
    let oa = r
        .occurrences
        .iter()
        .find(|x| x.occurrence == e.evidence.occurrence.to_record_identity().unwrap());
    assert_eq!(e.occurrence_attribution.as_ref(), oa);
    assert_eq!(
        e.contributor.as_ref(),
        oa.and_then(|x| x.contributor.as_ref())
            .and_then(|id| r.contributors.iter().find(|x| &x.id == id))
    );
    assert_eq!(
        e.distributor.as_ref(),
        snap.and_then(|x| x.distributor.as_ref())
            .and_then(|id| r.distributors.iter().find(|x| &x.id == id))
    );
}
